# 3-crash-sync-off-wait: 비동기 커밋 뒤 WAL writer의 한 주기(10초)를 기다린 다음 서버가 죽으면, 그 행은 살아남는가
# 3-crash-sync-off와 같고 kill 앞에 11초를 쉽니다. 쉬는 동안 WAL writer가 깨어 비동기 커밋 위치까지 쓰고 내리기를 기대합니다.
setup: CREATE TABLE acct(id int PRIMARY KEY, owner text, balance int);
setup: INSERT INTO acct VALUES (1, 'kim', 100);
setup: ALTER SYSTEM SET wal_writer_delay = '10s';
watch: acct
S1: SET synchronous_commit = off;
S1: INSERT INTO acct VALUES (2, 'lee', 200);
pause: 11s
kill: S1
S2: SELECT ctid, xmin, xmax, id, owner, balance FROM acct;
S2: INSERT INTO acct VALUES (3, 'park', 300);
S2: SELECT ctid, xmin, xmax, id, owner, balance FROM acct;
