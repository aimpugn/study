# 4-skew-rr: 당직 의사 둘이 REPEATABLE READ로 "당직이 둘이면 나는 빠진다"를 동시에 하기
# 결과를 보는 SELECT는 조건 없이 써서 모든 줄을 판정하게 합니다(실험 2).
setup: CREATE TABLE doctors(name text PRIMARY KEY, on_call bool);
setup: INSERT INTO doctors VALUES ('alice', true), ('bob', true);
watch: doctors
S1: BEGIN ISOLATION LEVEL REPEATABLE READ;
S1: SELECT count(*) FROM doctors WHERE on_call;
S2: BEGIN ISOLATION LEVEL REPEATABLE READ;
S2: SELECT count(*) FROM doctors WHERE on_call;
S1: UPDATE doctors SET on_call = false WHERE name = 'alice';
S2: UPDATE doctors SET on_call = false WHERE name = 'bob';
S1: COMMIT;
S2: COMMIT;
S1: SELECT ctid, xmin, xmax, name, on_call FROM doctors;
