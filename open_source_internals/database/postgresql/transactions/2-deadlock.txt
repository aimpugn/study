# 2-deadlock: 두 계좌 사이에서 서로 반대 순서로 이체하기
# 주의(2026-10-14 기록 뒤): 마지막 SELECT의 ORDER BY id는 기본 키 인덱스 스캔이 되어 HOT 사슬의 줄만 판정한다. 결과 SELECT는 순차 스캔이 되게 쓴다(조건 없이, 또는 인덱스 없는 열로). 기록과 맞추려고 이 파일은 그대로 둔다.
setup: CREATE TABLE acct(id int PRIMARY KEY, owner text, balance int);
setup: INSERT INTO acct VALUES (1, 'kim', 100), (2, 'lee', 200);
watch: acct
S1: BEGIN;
S1: UPDATE acct SET balance = balance - 10 WHERE id = 1;
S2: BEGIN;
S2: UPDATE acct SET balance = balance - 20 WHERE id = 2;
S1: UPDATE acct SET balance = balance + 10 WHERE id = 2;
S2: UPDATE acct SET balance = balance + 20 WHERE id = 1;
S1: ROLLBACK;
S2: COMMIT;
S2: SELECT ctid, xmin, xmax, id, balance FROM acct ORDER BY id;
